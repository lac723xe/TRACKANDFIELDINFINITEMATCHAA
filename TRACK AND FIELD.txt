local function jitter(base)
	return base * (0.8 + math.random() * 0.4)
end

local function panic()
	if sprintActive then sprintActive = false; stopSprint(); SprintToggle:SetValue(false) end
	if jumpActive then jumpActive = false; stopJump(); JumpToggle:SetValue(false) end
	if hopActive then hopActive = false; stopAutoHop(); HopToggle:SetValue(false) end
	if ljActive then ljActive = false; stopLongJump(); LJToggle:SetValue(false) end
	if clickerActive then clickerActive = false; stopAutoClicker(); ClickerToggle:SetValue(false) end
	if zoneActive then zoneActive = false; stopZoneProfiles(); ZoneToggle:SetValue(false) end
	notify("Panic", "All features disabled")
end

local function updateSprintKeys()
	sprintKey1VK = Key1Bind and (keyNameToVK[Key1Bind.Value] or sprintKey1VK) or sprintKey1VK
	sprintKey2VK = Key2Bind and (keyNameToVK[Key2Bind.Value] or sprintKey2VK) or sprintKey2VK
end

local function updateQuadKeys()
	sprintKey3VK = Key3Bind and (keyNameToVK[Key3Bind.Value] or sprintKey3VK) or sprintKey3VK
	sprintKey4VK = Key4Bind and (keyNameToVK[Key4Bind.Value] or sprintKey4VK) or sprintKey4VK
end

local function startSprint()
	if sprint_thread then return end
	sprintStop = false
	tapCount = 0; tapStart = tick()
	sprint_thread = task.spawn(function()
		local step = 1
		while sprintActive and not Library.Unloaded and not sprintStop do
			local maxStep = quadSprint and 4 or 2
			if step == 1 then
				keypress(sprintKey1VK)
				for _ = 1, holdTime do task.wait(jitter(0.016)) end
				keyrelease(sprintKey1VK)
			elseif step == 2 then
				keypress(sprintKey2VK)
				for _ = 1, holdTime do task.wait(jitter(0.016)) end
				keyrelease(sprintKey2VK)
			elseif step == 3 then
				keypress(sprintKey3VK)
				for _ = 1, holdTime do task.wait(jitter(0.016)) end
				keyrelease(sprintKey3VK)
			elseif step == 4 then
				keypress(sprintKey4VK)
				for _ = 1, holdTime do task.wait(jitter(0.016)) end
				keyrelease(sprintKey4VK)
			end
			step = step + 1
			if step > maxStep then step = 1 end
			tapCount = tapCount + 1
			task.wait(jitter(tapSpeed / 100))
		end
		sprint_thread = nil
	end)
end

local function startAutoJump()
	if jump_thread then return end
	jumpStop = false
	jump_thread = task.spawn(function()
		while jumpActive and not Library.Unloaded and not jumpStop do
			keypress(VK_SPACE)
			for _ = 1, holdTime do task.wait(jitter(0.016)) end
			keyrelease(VK_SPACE)
			task.wait(jitter(0.5))
		end
		jump_thread = nil
	end)
end

local function startAutoHop()
	if hop_thread then return end
	hopStop = false
	hop_thread = task.spawn(function()
		while hopActive and not Library.Unloaded and not hopStop do
			keypress(VK_SPACE)
			for _ = 1, holdTime do task.wait(jitter(0.016)) end
			keyrelease(VK_SPACE)
			task.wait(jitter(hopInterval))
		end
		hop_thread = nil
	end)
end

local function startLongJump()
	if lj_thread then return end
	ljStop = false; ljNotified = false
	lj_thread = task.spawn(function()
		while ljActive and not Library.Unloaded and not ljStop do
			if sprintActive then
				if not ljNotified and ljPoint > 10 then
					task.wait(jitter((ljPoint - 10) / 10))
					if ljActive and sprintActive then
						Library:Notify({ Title = "Long Jump", Content = "Jumping soon!", Duration = 2 })
						ljNotified = true
					end
					task.wait(jitter(10 / 10))
				else
					task.wait(jitter(ljPoint / 10))
				end
				if not (ljActive and sprintActive) then break end
				stopSprint()
				sprintActive = false
				SprintToggle:SetValue(false)
				keypress(VK_SPACE)
				for _ = 1, ljHold do task.wait(jitter(0.016)) end
				keyrelease(VK_SPACE)
			end
			task.wait(0.1)
		end
		lj_thread = nil
	end)
end

local function startAutoClicker()
	if clicker_thread then return end
	clickerStop = false
	clicker_thread = task.spawn(function()
		while clickerActive and not Library.Unloaded and not clickerStop do
			mouse1click()
			task.wait(jitter(clickerSpeed / 100))
		end
		clicker_thread = nil
	end)
end

local function startZoneProfiles()
	if zone_thread then return end
	zoneStop = false
	zone_thread = task.spawn(function()
		while zoneActive and not Library.Unloaded and not zoneStop do
			local char = Players.LocalPlayer and Players.LocalPlayer.Character
			local part = char and (char.PrimaryPart or char:FindFirstChild("HumanoidRootPart"))
			if part and sprintActive then
				if not startPos then startPos = part.Position end
				local dist = (part.Position - startPos).Magnitude
				if dist < 40 then
					tapSpeed = 22
				elseif dist < 100 then
					tapSpeed = 16
				else
					tapSpeed = 12
				end
			elseif not sprintActive then
				startPos = nil
			end
			task.wait(0.5)
		end
		zone_thread = nil; startPos = nil
	end)
end

local presetNames = { "100m", "200m", "300m", "400m", "4x100m Relay", "4x200m Relay", "800m", "1500m", "Hurdles", "Long Jump", "Throws" }

local function applyPresetData(data)
	if data.quadSprint ~= nil then quadSprint = data.quadSprint end
	if data.tapSpeed then SpeedSlider:SetValue(data.tapSpeed) end
	if data.holdTime then HoldSlider:SetValue(data.holdTime) end
	if data.hopInterval then HopSlider:SetValue(data.hopInterval * 10) end
	if data.ljPoint then LJPointSlider:SetValue(data.ljPoint) end
	if data.ljHold then LJHoldSlider:SetValue(data.ljHold) end
	if data.clickerSpeed then ClickerSlider:SetValue(data.clickerSpeed) end
	if data.jump then JumpToggle:SetValue(data.jump) end
	if data.hop then HopToggle:SetValue(data.hop) end
	if data.lj then LJToggle:SetValue(data.lj) end
	if data.clicker then ClickerToggle:SetValue(data.clicker) end
	if data.zone then ZoneToggle:SetValue(data.zone) end
end

local function collectPresetData()
	return {
		quadSprint = quadSprint,
		tapSpeed = tapSpeed,

		hop = hopActive,
		lj = ljActive,
		clicker = clickerActive,
		zone = zoneActive,
	}
end
